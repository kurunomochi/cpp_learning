cmake_minimum_required(VERSION 3.24)
project(cpp_foundations LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
set(lesson_names
  01_values 02_numbers 03_control 04_functions 05_references 06_containers
  07_classes 08_lifetime 09_algorithms 10_templates 11_errors 12_threads)
foreach(name IN LISTS lesson_names)
  add_executable(basic_${name} ${name}.cpp)
  target_compile_features(basic_${name} PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(basic_${name} PRIVATE /W4 /utf-8 /permissive-)
  else()
    target_compile_options(basic_${name} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(BUILD_TESTING)
    add_test(NAME basic_${name} COMMAND basic_${name})
  endif()
endforeach()
target_link_libraries(basic_12_threads PRIVATE Threads::Threads)
