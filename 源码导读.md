```cpp
struct Note { int id; std::string title; bool done{false}; };
template<std::integral T>
T twice(T value) { return value + value; }
int sum(std::span<const int> values) {
    int result{};
    for (int n : values) result += n;
    return result;
}
int main() {
    std::vector<Note> notes{{1, "Learn RAII", false}, {2, "Learn CMake", true}};
    auto found = std::ranges::find_if(notes, [](const Note& n) { return n.id == 1; });
    if (found != notes.end()) found->done = true;
    auto active = notes | std::views::filter([](const Note& n) { return n.done; });
    for (const auto& n : active) std::cout << n.title << '\n';
    auto owner = std::make_unique<Note>(Note{3, "Unique ownership"});
    auto moved = std::move(owner);
    if (owner || !moved) return 1;
    const int values[]{1, 2, 3};
    if (sum(values) != 6 || twice(4) != 8) return 2;
    int count{};
    std::mutex mutex;
    {
        std::vector<std::jthread> workers;
        for (int i = 0; i < 4; ++i) {
            workers.emplace_back([&] {
                for (int j = 0; j < 1000; ++j) {
                    std::lock_guard lock(mutex);
                    ++count;
                }
            });
        }
    } // jthread joins before count is examined.
    std::cout << "count=" << count << '\n';
    return count == 4000 ? 0 : 3;
}
```

## src/domain.cpp

先看 title_error 与 parse_id：领域规则是如何独立于 HTTP 和 SQLite 表达的。

```cpp
#include "task/domain.hpp"
#include <algorithm>
#include <charconv>
#include <stdexcept>
#include <system_error>

namespace task {
std::optional<std::string> title_error(std::string_view title) {
    if (title.empty() || title.size() > 200) return "title must contain 1..200 UTF-8 bytes";
    if (std::ranges::any_of(title, [](unsigned char c) { return c < 32 || c == 127; }))
        return "title must not contain ASCII control characters";
    if (std::ranges::all_of(title, [](char c) { return c == ' '; }))
        return "title must not contain only ASCII spaces";
    return std::nullopt;
}
std::optional<std::int64_t> parse_id(std::string_view text) {
    if (text.empty() || text.front() == '+' || text.front() == '-') return std::nullopt;
    std::int64_t value{};
    const auto [end, error] = std::from_chars(text.data(), text.data() + text.size(), value);
    if (error != std::errc{} || end != text.data() + text.size() || value <= 0) return std::nullopt;
    return value;
}
void validate_patch(const TaskPatch& patch) {
    if (!patch.title && !patch.completed) throw std::invalid_argument("patch must contain title or completed");
    if (patch.title) {
        if (auto error = title_error(*patch.title)) throw std::invalid_argument(*error);
    }
}
} // namespace task
```

```cpp
#include "task/domain.hpp"
#include <algorithm>
#include <iostream>
#include <sstream>
#include <string>
#include <vector>

int main() {
    std::vector<task::Task> tasks;
    std::int64_t next_id = 1;
    std::cout << "Commands: add TITLE | list | done ID | delete ID | quit\n";
    for (std::string line; std::cout << "> " && std::getline(std::cin, line);) {
        std::istringstream input(line);
        std::string command;
        input >> command;
        if (command == "quit") break;
        if (command == "add") {
            std::string title;
            std::getline(input >> std::ws, title);
            if (auto error = task::title_error(title)) { std::cout << *error << '\n'; continue; }
            tasks.push_back({next_id++, title, false});
            std::cout << "created " << tasks.back().id << '\n';
        } else if (command == "list") {
            for (const auto& t : tasks) std::cout << t.id << " [" << (t.completed ? 'x' : ' ') << "] " << t.title << '\n';
        } else if (command == "done" || command == "delete") {
            std::string raw, extra;
            input >> raw;
            auto id = task::parse_id(raw);
            if (!id || (input >> extra)) { std::cout << "invalid id\n"; continue; }
            auto it = std::ranges::find(tasks, *id, &task::Task::id);
            if (it == tasks.end()) { std::cout << "not found\n"; continue; }
            if (command == "done") it->completed = true; else tasks.erase(it);
            std::cout << "ok\n";
        } else if (!command.empty()) std::cout << "unknown command\n";
    }
}
```

## src/repository.cpp

按 Statement → 连接构造 → CRUD 的顺序读。逐一指出资源拥有者、参数绑定、锁范围和错误路径。

```cpp
#include "task/repository.hpp"
#include <sqlite3.h>
#include <utility>

namespace task {
namespace {
void check(sqlite3* db, int code) {
    if (code != SQLITE_OK) throw DatabaseError(code, sqlite3_errmsg(db));
}
class Statement {
public:
    Statement(sqlite3* db, const char* sql) : db_(db) {
        sqlite3_stmt* raw{};
        const int code = sqlite3_prepare_v2(db, sql, -1, &raw, nullptr);
        stmt_.reset(raw);
        check(db, code);
    }
    void bind(int index, const std::string& value) {
        check(db_, sqlite3_bind_text(stmt_.get(), index, value.data(),
                                     static_cast<int>(value.size()), SQLITE_TRANSIENT));
    }
    void bind(int index, std::int64_t value) { check(db_, sqlite3_bind_int64(stmt_.get(), index, value)); }
    void bind_null(int index) { check(db_, sqlite3_bind_null(stmt_.get(), index)); }
    bool row() {
        const int code = sqlite3_step(stmt_.get());
        if (code == SQLITE_ROW) return true;
        if (code == SQLITE_DONE) return false;
        throw DatabaseError(code, sqlite3_errmsg(db_));
    }
    Task task() const {
        const auto* text = sqlite3_column_text(stmt_.get(), 1);
        if (!text) throw DatabaseError(SQLITE_NOMEM, "cannot read task title");
        const auto size = sqlite3_column_bytes(stmt_.get(), 1);
        return {sqlite3_column_int64(stmt_.get(), 0),
                std::string(reinterpret_cast<const char*>(text), static_cast<std::size_t>(size)),
                sqlite3_column_int(stmt_.get(), 2) != 0};
    }
private:
    struct Finalize { void operator()(sqlite3_stmt* p) const noexcept { sqlite3_finalize(p); } };
    sqlite3* db_; // Non-owning; repository outlives every statement.
    std::unique_ptr<sqlite3_stmt, Finalize> stmt_;
};
void execute(sqlite3* db, const char* sql) { check(db, sqlite3_exec(db, sql, nullptr, nullptr, nullptr)); }
} // namespace

void DatabaseCloser::operator()(sqlite3* db) const noexcept { sqlite3_close_v2(db); }
TaskRepository::TaskRepository(const std::string& path) {
    sqlite3* raw{};
    const int code = sqlite3_open_v2(path.c_str(), &raw,
        SQLITE_OPEN_READWRITE | SQLITE_OPEN_CREATE | SQLITE_OPEN_FULLMUTEX, nullptr);
    db_.reset(raw); // Also closes a non-null handle when open failed.
    if (code != SQLITE_OK) throw DatabaseError(code, raw ? sqlite3_errmsg(raw) : "SQLite allocation failure");
    check(raw, sqlite3_busy_timeout(raw, 3000));
    execute(raw, "PRAGMA journal_mode=WAL;");
    execute(raw, "PRAGMA foreign_keys=ON;");
    execute(raw, "CREATE TABLE IF NOT EXISTS tasks("
        "id INTEGER PRIMARY KEY AUTOINCREMENT,"
        "title TEXT NOT NULL CHECK(length(CAST(title AS BLOB)) BETWEEN 1 AND 200),"
        "completed INTEGER NOT NULL DEFAULT 0 CHECK(completed IN (0,1))); ");
}
Task TaskRepository::create(const std::string& title) {
    if (auto error = title_error(title)) throw std::invalid_argument(*error);
    std::lock_guard lock(mutex_);
    Statement stmt(db_.get(), "INSERT INTO tasks(title) VALUES(?1) RETURNING id,title,completed;");
    stmt.bind(1, title);
    if (!stmt.row()) throw DatabaseError(SQLITE_ERROR, "insert returned no task");
    auto result = stmt.task();
    (void)stmt.row(); // Finish the write before returning; commit failures surface here.
    return result;
}
std::vector<Task> TaskRepository::list(int limit, int offset) {
    if (limit < 1 || limit > 100 || offset < 0 || offset > 1000000)
        throw std::invalid_argument("invalid pagination");
    std::lock_guard lock(mutex_);
    Statement stmt(db_.get(), "SELECT id,title,completed FROM tasks ORDER BY id LIMIT ?1 OFFSET ?2;");
    stmt.bind(1, static_cast<std::int64_t>(limit));
    stmt.bind(2, static_cast<std::int64_t>(offset));
    std::vector<Task> result;
    while (stmt.row()) result.push_back(stmt.task());
    return result;
}
std::optional<Task> TaskRepository::get(std::int64_t id) {
    std::lock_guard lock(mutex_);
    Statement stmt(db_.get(), "SELECT id,title,completed FROM tasks WHERE id=?1;");
    stmt.bind(1, id);
    if (!stmt.row()) return std::nullopt;
    return stmt.task();
}
std::optional<Task> TaskRepository::update(std::int64_t id, const TaskPatch& patch) {
    validate_patch(patch);
    std::lock_guard lock(mutex_);
    Statement stmt(db_.get(), "UPDATE tasks SET title=COALESCE(?1,title),"
        "completed=COALESCE(?2,completed) WHERE id=?3 RETURNING id,title,completed;");
    if (patch.title) stmt.bind(1, *patch.title); else stmt.bind_null(1);
    if (patch.completed) stmt.bind(2, static_cast<std::int64_t>(*patch.completed)); else stmt.bind_null(2);
    stmt.bind(3, id);
    if (!stmt.row()) return std::nullopt;
    auto result = stmt.task();
    (void)stmt.row();
    return result;
}
bool TaskRepository::erase(std::int64_t id) {
    std::lock_guard lock(mutex_);
    Statement stmt(db_.get(), "DELETE FROM tasks WHERE id=?1;");
    stmt.bind(1, id);
    (void)stmt.row();
    return sqlite3_changes(db_.get()) == 1;
}
} // namespace task
```

## src/http_api.cpp

先读 respond/fail/body_object 等辅助函数，再读 register_routes。观察协议错误与数据库错误怎样映射为状态码。

```cpp
#include "task/http_api.hpp"
#include "task/repository.hpp"
#include <httplib.h>
#include <nlohmann/json.hpp>
#include <algorithm>
#include <atomic>
#include <charconv>
#include <cctype>
#include <iostream>
#include <memory>
#include <syncstream>

namespace task {
namespace {
using Json = nlohmann::json;
class HttpError : public std::runtime_error {
public:
    HttpError(int status_value, std::string code_value, std::string message)
        : std::runtime_error(std::move(message)), status(status_value), code(std::move(code_value)) {}
    int status;
    std::string code;
};
void respond(httplib::Response& response, int status, const Json& body) {
    response.status = status;
    response.set_content(body.dump(), "application/json; charset=utf-8");
}
void fail(httplib::Response& response, int status, const std::string& code, const std::string& message) {
    respond(response, status, {{"error", {{"code", code}, {"message", message}}}});
}
Json as_json(const Task& value) {
    return {{"id", value.id}, {"title", value.title}, {"completed", value.completed}};
}
Json body_object(const httplib::Request& request) {
    auto type = request.get_header_value("Content-Type");
    type = type.substr(0, type.find(';'));
    while (!type.empty() && type.back() == ' ') type.pop_back();
    std::ranges::transform(type, type.begin(), [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
    if (type != "application/json") throw HttpError(415, "unsupported_media_type", "Content-Type must be application/json");
    auto body = Json::parse(request.body, nullptr, false);
    if (body.is_discarded()) throw HttpError(400, "invalid_json", "request body is not valid JSON");
    if (!body.is_object()) throw HttpError(422, "validation_error", "request body must be a JSON object");
    return body;
}
std::string read_title(const Json& body) {
    if (!body.contains("title") || !body.at("title").is_string())
        throw HttpError(422, "validation_error", "title must be a string");
    auto title = body.at("title").get<std::string>();
    if (auto error = title_error(title)) throw HttpError(422, "validation_error", *error);
    return title;
}
std::int64_t request_id(const httplib::Request& request) {
    auto id = parse_id(request.path_params.at("id"));
    if (!id) throw HttpError(400, "invalid_id", "id must be a positive 64-bit integer");
    return *id;
}
int query_int(const httplib::Request& request, const char* name, int fallback, int minimum, int maximum) {
    if (!request.has_param(name)) return fallback;
    const auto text = request.get_param_value(name);
    int result{};
    const auto [end, error] = std::from_chars(text.data(), text.data() + text.size(), result);
    if (request.get_param_value_count(name) != 1 || error != std::errc{} ||
        end != text.data() + text.size() || result < minimum || result > maximum)
        throw HttpError(400, "invalid_query", std::string(name) + " is out of range or not an integer");
    return result;
}
} // namespace

void register_routes(httplib::Server& server, TaskRepository& repository, std::string api_token) {
    auto sequence = std::make_shared<std::atomic<std::uint64_t>>(0);
    server.set_pre_routing_handler([token = std::move(api_token), sequence](const auto& request, auto& response) {
        response.set_header("X-Request-Id", std::to_string(sequence->fetch_add(1, std::memory_order_relaxed) + 1));
        response.set_header("Cache-Control", "no-store");
        if (request.path.starts_with("/api/") && !token.empty() &&
            request.get_header_value("Authorization") != "Bearer " + token) {
            response.set_header("WWW-Authenticate", "Bearer");
            fail(response, 401, "unauthorized", "a valid bearer token is required");
            return httplib::Server::HandlerResponse::Handled;
        }
        return httplib::Server::HandlerResponse::Unhandled;
    });
    server.set_exception_handler([](const auto&, auto& response, std::exception_ptr error) {
        try {
            std::rethrow_exception(error);
        } catch (const HttpError& e) {
            fail(response, e.status, e.code, e.what());
        } catch (const DatabaseError& e) {
            // SQLite primary error codes: BUSY=5, LOCKED=6.
            const bool busy = (e.code() & 0xff) == 5 || (e.code() & 0xff) == 6;
            fail(response, busy ? 503 : 500, busy ? "database_busy" : "internal_error",
                 busy ? "database is busy; retry later" : "an internal error occurred");
            std::osyncstream(std::cerr) << Json{{"event", "database_error"}, {"sqlite_code", e.code()}}.dump() << '\n';
        } catch (const std::exception&) {
            fail(response, 500, "internal_error", "an internal error occurred");
            std::osyncstream(std::cerr) << "{\"event\":\"unhandled_exception\"}\n";
        } catch (...) {
            fail(response, 500, "internal_error", "an internal error occurred");
        }
    });
    server.set_error_handler([](const auto&, auto& response) {
        if (response.body.empty()) fail(response, response.status, "http_error", "request could not be handled");
    });
    server.set_logger([](const auto& request, const auto& response) {
        // JSON escaping prevents log injection. Never log token or request body.
        std::osyncstream(std::cout) << Json{{"method", request.method}, {"path", request.path},
            {"status", response.status}, {"request_id", response.get_header_value("X-Request-Id")}}.dump() << '\n';
    });
    server.Get("/healthz", [](const auto&, auto& response) { respond(response, 200, {{"status", "ok"}}); });
    server.Get("/api/tasks", [&repository](const auto& request, auto& response) {
        const int limit = query_int(request, "limit", 20, 1, 100);
        const int offset = query_int(request, "offset", 0, 0, 1000000);
        auto items = Json::array();
        for (const auto& value : repository.list(limit, offset)) items.push_back(as_json(value));
        respond(response, 200, {{"items", items}, {"limit", limit}, {"offset", offset}});
    });
    server.Post("/api/tasks", [&repository](const auto& request, auto& response) {
        const Json body = body_object(request);
        if (body.size() != 1 || !body.contains("title"))
            throw HttpError(422, "validation_error", "creation accepts exactly one field: title");
        const auto value = repository.create(read_title(body));
        response.set_header("Location", "/api/tasks/" + std::to_string(value.id));
        respond(response, 201, as_json(value));
    });
    server.Get("/api/tasks/:id", [&repository](const auto& request, auto& response) {
        auto value = repository.get(request_id(request));
        if (!value) throw HttpError(404, "not_found", "task does not exist");
        respond(response, 200, as_json(*value));
    });
    server.Patch("/api/tasks/:id", [&repository](const auto& request, auto& response) {
        const auto id = request_id(request);
        const Json body = body_object(request);
        TaskPatch patch;
        if (body.empty()) throw HttpError(422, "validation_error", "patch must contain title or completed");
        for (auto it = body.begin(); it != body.end(); ++it) {
            if (it.key() != "title" && it.key() != "completed")
                throw HttpError(422, "validation_error", "unknown patch field");
        }
        if (body.contains("title")) patch.title = read_title(body);
        if (body.contains("completed")) {
            if (!body.at("completed").is_boolean()) throw HttpError(422, "validation_error", "completed must be boolean");
            patch.completed = body.at("completed").get<bool>();
        }
        auto value = repository.update(id, patch);
        if (!value) throw HttpError(404, "not_found", "task does not exist");
        respond(response, 200, as_json(*value));
    });
    server.Delete("/api/tasks/:id", [&repository](const auto& request, auto& response) {
        if (!repository.erase(request_id(request))) throw HttpError(404, "not_found", "task does not exist");
        response.status = 204;
    });
}
} // namespace task
```

## src/main.cpp

配置解析 → 对象构造 → 资源上限 → 路由注册 → 信号与退出 → 监听。注意局部对象的反向析构顺序。

```cpp
#include "task/http_api.hpp"
#include "task/repository.hpp"
#include <httplib.h>
#include <atomic>
#include <chrono>
#include <csignal>
#include <cstdlib>
#include <iostream>
#include <stdexcept>
#include <string>
#include <thread>

namespace {
std::atomic<bool> stopping{false};
static_assert(std::atomic<bool>::is_always_lock_free);
extern "C" void handle_signal(int) { stopping.store(true, std::memory_order_relaxed); }
std::string environment(const char* name, const char* fallback) {
    const char* value = std::getenv(name);
    return value && *value ? value : fallback;
}
}
int main() {
    try {
        const auto host = environment("APP_HOST", "127.0.0.1");
        const auto port_value = task::parse_id(environment("APP_PORT", "8080"));
        if (!port_value || *port_value > 65535) throw std::invalid_argument("APP_PORT must be 1..65535");
        const auto path = environment("APP_DB", "tasks.db");
        const auto token = environment("APP_TOKEN", "");
        if (host != "127.0.0.1" && host != "localhost" && host != "::1" && token.size() < 32)
            throw std::invalid_argument("non-loopback APP_HOST requires APP_TOKEN of at least 32 bytes");
        task::TaskRepository repository(path);
        httplib::Server server;
        server.new_task_queue = [] { return new httplib::ThreadPool(4, 4, 64); };
        server.set_payload_max_length(16 * 1024);
        server.set_read_timeout(std::chrono::seconds(5));
        server.set_write_timeout(std::chrono::seconds(5));
        server.set_keep_alive_max_count(20);
        task::register_routes(server, repository, token);
        std::signal(SIGINT, handle_signal);
        std::signal(SIGTERM, handle_signal);
        if (!server.bind_to_port(host, static_cast<int>(*port_value))) {
            std::cerr << "Cannot bind server address; check APP_HOST and APP_PORT\n";
            return 1;
        }
        std::jthread shutdown_watcher([&server](std::stop_token stop) {
            while (!stop.stop_requested()) {
                if (stopping.load(std::memory_order_relaxed) && server.is_running()) { server.stop(); return; }
                std::this_thread::sleep_for(std::chrono::milliseconds(100));
            }
        });
        std::cout << "Task server: http://" << host << ':' << *port_value
                  << "  (Ctrl+C to stop)\n" << std::flush;
        const bool ok = server.listen_after_bind();
        return ok ? 0 : 1;
    } catch (const std::exception& error) {
        std::cerr << "Startup failed: " << error.what() << '\n';
        return 1;
    }
}
```

```python
"""Black-box integration tests. Standard library only; never touch a user's DB."""
import concurrent.futures
import json
import os
from pathlib import Path
import socket
import sqlite3
import subprocess
import sys
import tempfile
import time
import urllib.error
import urllib.request

TOKEN = "integration-test-token-012345678901234567890123456789"
checks = 0


def require(condition, message):
    global checks
    checks += 1
    if not condition:
        raise RuntimeError(message)


def request(port, method, path, body=None, token=TOKEN, content_type="application/json", raw=None):
    headers = {"Content-Type": content_type}
    if token:
        headers["Authorization"] = "Bearer " + token
    data = raw if raw is not None else (json.dumps(body, ensure_ascii=False).encode("utf-8") if body is not None else None)
    req = urllib.request.Request(f"http://127.0.0.1:{port}{path}", data=data, headers=headers, method=method)
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))
    try:
        response = opener.open(req, timeout=10)
    except urllib.error.HTTPError as error:
        response = error
    with response:
        text = response.read().decode("utf-8")
        return response.status, (json.loads(text) if text else None), dict(response.headers)


def launch(binary, directory, port):
    env = os.environ.copy()
    env.update(APP_HOST="127.0.0.1", APP_PORT=str(port), APP_DB=str(directory / "test.db"), APP_TOKEN=TOKEN)
    log = open(directory / "server.log", "ab", buffering=0)
    process = subprocess.Popen([binary], env=env, cwd=directory, stdout=log, stderr=log)
    try:
        deadline = time.monotonic() + 15
        while time.monotonic() < deadline:
            if process.poll() is not None:
                raise RuntimeError("server exited during startup: " + (directory / "server.log").read_text(errors="replace"))
            try:
                if request(port, "GET", "/healthz")[0] == 200:
                    return process, log
            except (OSError, urllib.error.URLError):
                pass
            time.sleep(0.05)
        raise RuntimeError("server readiness timeout")
    except BaseException:
        stop(process, log)
        raise


def stop(process, log):
    if process.poll() is None:
        process.terminate()  # Windows: abrupt termination; POSIX: SIGTERM.
        try:
            process.wait(timeout=10)
        except subprocess.TimeoutExpired:
            process.kill()
            process.wait(timeout=5)
    log.close()

        directory = Path(tmp)

            require(status == 201 and first["title"] == "学习 C++", "create UTF-8 task")
            task_id = first["id"]
            require(headers.get("Location") == f"/api/tasks/{task_id}", "Location header")
            require(bool(headers.get("X-Request-Id")), "request ID header")
            route = f"/api/tasks/{task_id}"
            require(request(port, "GET", route)[1] == first, "read task")
            require(request(port, "PATCH", route, {"completed": True})[1]["completed"] is True, "complete task")
            require(request(port, "PATCH", route, {"completed": False})[1]["completed"] is False, "uncomplete task")
            require(request(port, "PATCH", route, {"title": "renamed"})[1]["title"] == "renamed", "rename task")
            for data in ({}, {"title": ""}, {"title": " "}, {"title": 7}, {"title": "a" * 201}, {"title": "a\nb"}, {"title": "ok", "admin": True}, []):
                require(request(port, "POST", "/api/tasks", data)[0] == 422, f"reject invalid creation {data!r}")
            for data in ({}, {"completed": "false"}, {"completed": None}, {"extra": True}):
                require(request(port, "PATCH", route, data)[0] == 422, f"reject invalid patch {data!r}")
            require(request(port, "POST", "/api/tasks", raw=b"{")[0] == 400, "malformed JSON")
            require(request(port, "POST", "/api/tasks", {"title": "ok"}, content_type="text/plain")[0] == 415, "media type")
            require(request(port, "POST", "/api/tasks", {"title": "x" * 17000})[0] == 413, "body size bound")
            for suffix in ("0", "-1", "abc", "99999999999999999999999999"):
                require(request(port, "GET", "/api/tasks/" + suffix)[0] == 400, "invalid path ID")
            for query in ("limit=0", "limit=101", "offset=-1", "offset=1000001", "limit=1x", "limit=1&limit=2"):
                require(request(port, "GET", "/api/tasks?" + query)[0] == 400, "invalid query")
            require(request(port, "GET", "/missing")[0] == 404, "unknown route")
            require(request(port, "GET", "/api/tasks/999999")[0] == 404, "missing resource")
            require(request(port, "PATCH", "/api/tasks/999999", {"completed": True})[0] == 404, "update missing resource")
            require(request(port, "POST", "/api/tasks", {"title": "a" * 200})[0] == 201, "200-byte title accepted")
            require(request(port, "POST", "/api/tasks", {"title": "学" * 67})[0] == 422, "UTF-8 length uses bytes")
            require(request(port, "POST", "/api/tasks", raw=b'{"title":"\xff"}')[0] == 400, "invalid UTF-8 rejected")
```

## scripts/verify_lessons.py

对应文件可直接在编辑器中打开；修改后按 README 重新构建并运行相关测试。

```python
"""Build the teaching projects and check their documented behaviors (Python 3.9+)."""
import argparse
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--build-root', type=Path, help='Scratch directory for builds and logs')
    args = parser.parse_args()
    source = Path(__file__).resolve().parents[1]
    output = (args.build_root or source / 'build/lesson-verification').resolve()
    output.mkdir(parents=True, exist_ok=True)
    results = []

    def run(label, command, expected=0, contains=None):
        print(label, flush=True)
        process = subprocess.run([str(x) for x in command], stdout=subprocess.PIPE,
                                 stderr=subprocess.STDOUT, encoding='utf-8', errors='replace',
                                 timeout=240, check=False)
        (output / (label + '.txt')).write_text(process.stdout, encoding='utf-8')
        ok = process.returncode == expected if expected == 0 else process.returncode != 0
        if contains is not None:
            ok = ok and contains in process.stdout
        results.append({'name': label, 'passed': ok, 'exit_code': process.returncode,
                        'expected_failure': expected != 0})
        (output / 'results.json').write_text(json.dumps(results, ensure_ascii=False, indent=2), encoding='utf-8')
        if not ok:
            raise RuntimeError(f'{label} failed:\n{process.stdout}')
        return process.stdout

    def configure(name, directory, config, *options, generator='Ninja'):
        destination = output / name
        command = ['cmake', '-S', source / directory, '-B', destination, '-G', generator]
        if generator == 'Ninja':
            command.append('-DCMAKE_BUILD_TYPE=' + config)
        command.extend(options)
        run(name + '-configure', command)
        return destination

    def build(name, directory, config):
        run(name + '-build', ['cmake', '--build', directory, '--config', config, '--parallel', '2'])

    def test(name, directory, config, expected=0, contains=None):
        return run(name + '-test', ['ctest', '--test-dir', directory, '-C', config,
                                    '--output-on-failure', '--no-tests=error'], expected, contains)

    def executable(directory, name):
        return directory / (name + ('.exe' if os.name == 'nt' else ''))

    run('cmake-version', ['cmake', '--version'])
    run('ninja-version', ['ninja', '--version'])
    run('variable-scope', ['cmake', '-P', source / 'lessons/cmake/variables.cmake'],
        contains='outside label=outer')
    for config in ('Debug', 'Release'):
        suffix = config.lower()
        name = 'foundations-' + suffix
        directory = configure(name, 'lessons/foundations', config)
        build(name, directory, config)
        test(name, directory, config, contains='12/12')
        trace = run(name + '-lifetime', [executable(directory, 'basic_08_lifetime')])
        expected_trace = ['open outer', 'open resource', 'owner moved', 'close resource',
                          'leaving main', 'close outer']
        if trace.splitlines() != expected_trace:
            raise RuntimeError('RAII destruction order differs from the lesson')

        name = 'hello-' + suffix
        directory = configure(name, 'lessons/cmake/01_hello', config)
        build(name, directory, config)
        run(name + '-run', [executable(directory, 'hello')], contains='Hello, CMake!')

        name = 'library-' + suffix
        directory = configure(name, 'lessons/cmake/02_library', config)
        build(name, directory, config)
        test(name, directory, config)

        name = 'taskkit-' + suffix
        directory = configure(name, 'lessons/cmake/03_project', config)
        build(name, directory, config)
        test(name, directory, config)
        run(name + '-run', [executable(directory / 'app', 'taskkit_app')], contains='TaskKit 1.0.0\ntrue')
        prefix = output / ('install-' + suffix)
        run(name + '-install', ['cmake', '--install', directory, '--config', config, '--prefix', prefix])

        # Two independent consumer builds: original install tree and copied install tree.
        for mode in ('original', 'relocated'):
            consumer_prefix = prefix
            if mode == 'relocated':
                consumer_prefix = output / ('relocated-' + suffix)
                shutil.copytree(prefix, consumer_prefix, dirs_exist_ok=True)
            name = 'consumer-' + mode + '-' + suffix
            directory = configure(name, 'lessons/cmake/04_consumer', config,
                                  '-DCMAKE_PREFIX_PATH=' + str(consumer_prefix))
            build(name, directory, config)
            test(name, directory, config)
            cache = (directory / 'CMakeCache.txt').read_text(encoding='utf-8')
            package_dir = next(line.split('=', 1)[1] for line in cache.splitlines()
                               if line.startswith('TaskKit_DIR:'))
            Path(package_dir).resolve().relative_to(consumer_prefix.resolve())

    name = 'taskkit-multiconfig'
    directory = configure(name, 'lessons/cmake/03_project', 'Debug', generator='Ninja Multi-Config')
    build(name, directory, 'Debug')
    test(name, directory, 'Debug')

    name = 'taskkit-library-only'
    directory = configure(name, 'lessons/cmake/03_project', 'Release',
                          '-DTASKKIT_BUILD_APP=OFF', '-DBUILD_TESTING=OFF')
    build(name, directory, 'Release')
    test(name, directory, 'Release', expected=1, contains='No tests were found')

    # Mutate ONLY an isolated copy to prove the boundary test really detects a bug.
    fault_source = output / 'fault-source'
    shutil.copytree(source / 'lessons/cmake/03_project', fault_source, dirs_exist_ok=True,
                    ignore=shutil.ignore_patterns('build', '.git', 'CMakeUserPresets.json'))
    cpp = fault_source / 'src/title.cpp'
    original = cpp.read_text(encoding='utf-8')
    if 'text.size() > 200' not in original:
        raise RuntimeError('Boundary mutation no longer matches the example')
    cpp.write_text(original.replace('text.size() > 200', 'text.size() > 199'), encoding='utf-8')
    fault_build = output / 'fault-build'
    run('negative-boundary-configure', ['cmake', '-S', fault_source, '-B', fault_build,
                                      '-G', 'Ninja', '-DCMAKE_BUILD_TYPE=Release'])
    build('negative-boundary', fault_build, 'Release')
    test('negative-boundary', fault_build, 'Release', expected=1, contains='FAIL: 200-byte boundary')
    print(f'PASS: {len(results)} verification steps; logs: {output}')


if __name__ == '__main__':
    try:
        main()
    except (RuntimeError, OSError, ValueError, StopIteration, subprocess.TimeoutExpired) as error:
        print(error, file=sys.stderr)
        sys.exit(1)
```
